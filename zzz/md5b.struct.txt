# string struct
string {
     length: int32
     characters: [length]char8
}

# md5b file struct
# md5mesh version 11
# https://github.com/XeNTaXBackup/XeNTaXBackup.github.io/blob/2c6074a1bcb085e77f527fcf620620a4ee1e76da/markdown/Quake%20Wars%20models_2802.md#L10
md5b {
    # header
    majorVersion: int32 = 1
    minorVersion: int32 = 1

    # joints
    numJoints: int32
    joints: [numJoints]Joint_t
        Joint_t { # not same as ASCII md5mesh, rotation and position data are local/relative, but are absolute in ASCII version
            jointName: string
            parentIndex: int32
            rotationQuat.x: float32 # idJointQuat::q::x
            rotationQuat.y: float32 # idJointQuat::q::y
            rotationQuat.z: float32 # idJointQuat::q::z
            rotationQuat.w: float32 # idJointQuat::q::w
            position.x: float32 # idJointQuat::t::x
            position.y: float32 # idJointQuat::t::y
            position.z: float32 # idJointQuat::t::z
            w: float32 # idJointQuat::w
            poseMat: [12]float32 # idJointMat::m # global/absolute transform
        }

    # meshes
    numMeshes: int32
    meshes: [numMeshes]Mesh_t
        Mesh_t {
            meshName: string
            majorIndex: int32
            minorIndex: int32
            unknown1: int32
            numRawVertex: int32 ???
            noAnimate: bool
            noAnimate: bool
            singleJointsOnVertex: bool # 1 = only has 1 joint on every vertex; 0 = has more than 1 joints on every vertex;
            shader: string
            numTexCoords: int32 # same as numVerts in ASCII version, so I named it `numTexCoords`
            numTriVertexes: int32 # numTris == numTriVertexes / 3; it is num of vertex, not num of triangle
            tris: [numTriVertexes]uint16
            numSilEdges: int32 # it seams is deformInfo_t under this
            silEdges: [numSilEdges * 4]uint16
            silIndexes: [numTexCoords]uint16
            numSourceVerts: int32
            numOutputVerts: int32
            numMirroredVerts: int32
            mirroredVerts?: [numMirroredVerts]int32
            unknown2: int32
            unknown3: int32
            numJointTables: int32
            jointIndices: [numJointTables]uint8
            drawVerts: [numOutputVerts]DrawVert_t
                DrawVert_t {
                    position.x: float32
                    position.y: float32
                    position.z: float32
                    texCoord.s: float32
                    texCoord.t: float32
                    normal.x: float32
                    normal.y: float32
                    normal.z: float32
                    tangent.x: float32
                    tangent.y: float32
                    tangent.z: float32
                    biTangentSign: float32
                    color.r: uint8
                    color.g: uint8
                    color.b: uint8
                    color.a: uint8
                }
            @if singleJointsOnVertex == 1: # vertex only 1 joint
            vertJointIndices: [numVertices]uint8 # VALUE / 3 == index of `jointIndices`, every weight is 1.0(255) on a vertex
            @elseif singleJointsOnVertex == 4: # has more joints and weights on a vertex
            vertJointIndices: [numVertices]VertJointIndex_t
                VertJointIndex_t {
                    jointIndex: uint8 # VALUE / 3 == index of `jointIndices`
                    weight: uint8 # VALUE / 255 == float joint weight, no more weights if equals 0
                }
            @endif
            unknown4: bool
            unknown5: float32
        }
        ...
    <...> # gui model surfaces
    <EOF>
}


struct idMD5Mesh {
	void *vtbl; // 0 + 4
	
	idStr name; // 4 + 32
	
	int index0_v36; // 36 + 4
	int index1_v40; // 40 + 4
	int v44; // 44 + 4
	idMaterial *shader_v48; // 48 + 4
	int v52; // 52 + 4
	
	char padding_1[16]; // 56 + 16
	
	int v72; // int* + 18 // 72 + 6
	int v76; // 76 + 4
	int numTris_v80; // int* + 20 // 80 + 4
	unsigned short *tris_v84; // length = v80 // 84 + 4
	
	char padding_2[4]; // 88 + 4
	
	int numSilEdges_v92; // int* + 23 // 92 + 4
	int *silEdges_v96; // 96 + 100
	
	char padding_3[8]; // 100 + 8
	
	short *v108_list_length_of_v72; // length = v72 * 2 // 108 + 4
	
	char padding_4[8]; // 112 + 8
	
	int numSourceVerts_v120; // 120 + 4
	int numOutputVerts_v124; // int* + 31 // 124 + 4
	int numMirroredVerts_v128; // 128 + 4
	
	int *mirroredVerts_v132; // length = v128 // int* + 33 // 132 + 4
	int v136; // 136 + 4
	char padding_5[8]; // 140 + 8
	int v148; // 148 + 4
	
	char padding_6[12]; // 152 + 12
	
	idList_byte jointIndex_v164; // int* + 41 = 16 // 164 + 16
	byte *v180; // int* + 45 length = 8 * v124 // 180 + 4
	
	idDrawVert *verts_v184; // int* + 46, length = v124 // 184 + 4
	
	char padding_7[8]; // 188 + 8
	
	bool bool_v196; // 196 + 1
	char padding_8[1]; // 197 + 1
	bool bool_v198; // 198 + 1
	
	char padding_9[17]; // 199 + 17
	
	bool bool_v216; // 216 + 1
	bool vertexRigidFlag_v217; // 217 + 1
	
	char padding_10[2]; // 218 + 2
	
	float v220; // 220 + 4
}
